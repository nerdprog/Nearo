-- 1. Create POSTS table
CREATE TABLE IF NOT EXISTS public.posts (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
  user_id text NOT NULL, -- Firebase UID
  username text NOT NULL,
  pfp_url text, -- User's chosen profile picture
  zone_id text NOT NULL,
  lat double precision,
  lng double precision,
  title text NOT NULL,
  content text NOT NULL,
  image_url text, -- Attached image URL (max 1 per day)
  upvotes integer DEFAULT 0,
  downvotes integer DEFAULT 0,
  vote_score integer DEFAULT 0, -- NEW: (upvotes - downvotes) for sorting
  flagged boolean DEFAULT false
);

-- Trigger to update vote_score
CREATE OR REPLACE FUNCTION update_vote_score()
RETURNS TRIGGER AS $$
BEGIN
  NEW.vote_score := NEW.upvotes - NEW.downvotes;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_update_vote_score
BEFORE INSERT OR UPDATE ON public.posts
FOR EACH ROW EXECUTE FUNCTION update_vote_score();

-- Enable RLS on POSTS
ALTER TABLE public.posts ENABLE ROW LEVEL SECURITY;

-- Post Policies
DROP POLICY IF EXISTS "Anyone can read posts" ON public.posts;
CREATE POLICY "Anyone can read posts"
ON public.posts FOR SELECT
USING (true);

DROP POLICY IF EXISTS "Users can create posts" ON public.posts;
CREATE POLICY "Users can create posts"
ON public.posts FOR INSERT
WITH CHECK (true);

DROP POLICY IF EXISTS "Users can delete their own posts" ON public.posts;
CREATE POLICY "Users can delete their own posts"
ON public.posts FOR DELETE
USING (user_id = current_setting('request.jwt.claims', true)::json->>'sub' OR user_id = current_setting('request.headers', true)::json->>'x-user-id');

DROP POLICY IF EXISTS "Users can update post votes" ON public.posts;
CREATE POLICY "Users can update post votes"
ON public.posts FOR UPDATE
USING (true);


-- 2. Create VOTES table
CREATE TABLE IF NOT EXISTS public.votes (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
  user_id text NOT NULL,
  post_id uuid REFERENCES public.posts(id) ON DELETE CASCADE,
  vote_type text CHECK (vote_type IN ('up', 'down'))
);

-- Prevent duplicate votes
CREATE UNIQUE INDEX IF NOT EXISTS unique_user_post_vote ON public.votes (user_id, post_id);

-- Enable RLS on VOTES
ALTER TABLE public.votes ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Users can manage their own votes" ON public.votes;
CREATE POLICY "Users can manage their own votes"
ON public.votes FOR ALL
USING (true)
WITH CHECK (true);


-- 3. Create REPORTS table
CREATE TABLE IF NOT EXISTS public.reports (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
  user_id text NOT NULL,
  post_id uuid REFERENCES public.posts(id) ON DELETE CASCADE
);

CREATE UNIQUE INDEX IF NOT EXISTS unique_user_post_report ON public.reports (user_id, post_id);

ALTER TABLE public.reports ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Users can insert reports" ON public.reports;
CREATE POLICY "Users can insert reports"
ON public.reports FOR INSERT
WITH CHECK (true);

DROP POLICY IF EXISTS "Users can view their own reports" ON public.reports;
CREATE POLICY "Users can view their own reports"
ON public.reports FOR SELECT
USING (true);


-- 4. Create STREAKS table
CREATE TABLE IF NOT EXISTS public.streaks (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id text UNIQUE NOT NULL,
  current_streak integer DEFAULT 0,
  last_post_date date
);

ALTER TABLE public.streaks ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Users can manage their own streaks" ON public.streaks;
CREATE POLICY "Users can manage their own streaks"
ON public.streaks FOR ALL
USING (true)
WITH CHECK (true);


-- 5. Create USER_DAILY_STATS table (NEW)
-- This tracks posts per day and image uploads per day to enforce limits securely.
CREATE TABLE IF NOT EXISTS public.user_daily_stats (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id text NOT NULL,
  stat_date date NOT NULL DEFAULT CURRENT_DATE,
  posts_count integer DEFAULT 0,
  has_image_uploaded boolean DEFAULT false
);

CREATE UNIQUE INDEX IF NOT EXISTS unique_user_date_stats ON public.user_daily_stats (user_id, stat_date);

ALTER TABLE public.user_daily_stats ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Users can manage their own stats" ON public.user_daily_stats;
CREATE POLICY "Users can manage their own stats"
ON public.user_daily_stats FOR ALL
USING (true)
WITH CHECK (true);

-- ==========================================
-- DAILY CLEANUP CRON JOB (Optional but recommended)
-- ==========================================
-- Automatically delete posts older than 24 hours
-- Requires pg_cron extension
SELECT cron.schedule('delete_old_posts', '0 0 * * *', $$
DELETE FROM public.posts WHERE created_at < NOW() - INTERVAL '24 hours';
$$);
